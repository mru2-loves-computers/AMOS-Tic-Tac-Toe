WIDTH=320
HEIGHT=256
Screen Open 0,WIDTH,HEIGHT,16,Lowres
Flash Off

Reserve Zone 9

Dim SPACES(9) : Rem:spaces empty=0 taken=1

Dim WIN(8,3)
'winning rows
WIN(0,0)=0 : WIN(0,1)=1 : WIN(0,2)=2
WIN(1,0)=3 : WIN(1,1)=4 : WIN(1,2)=5
WIN(2,0)=6 : WIN(2,1)=7 : WIN(2,2)=8

'winning columns
WIN(3,0)=0 : WIN(3,1)=3 : WIN(3,2)=6
WIN(4,0)=1 : WIN(4,1)=4 : WIN(4,2)=7
WIN(5,0)=2 : WIN(5,1)=5 : WIN(5,2)=8

'winning crosses
WIN(6,0)=0 : WIN(6,1)=4 : WIN(6,2)=8
WIN(7,0)=2 : WIN(7,1)=4 : WIN(7,2)=6

Gosub SETUP_GAME

Do
  If T>1 Then T=0
  
  If T=0
    If Mouse Click and Mouse Key=1
      Z=Mouse Zone
      Gosub CHECK_ZONE
      
      Gosub CHECK_WIN
    End If
  End If
  
  If T=1
    Gosub OPPONENT_MOVE
    T=0
    
    Gosub CHECK_WIN
  End If
  
  Wait Vbl
Loop

SETUP_GAME:
  Cls 0

  For I=0 To 8
    X=50+(I mod 3)*52
    Y=50+(I/3)*52
    Ink 9 : Bar X,Y To X+50,Y+50
    Set Zone I+1,X,Y To X+50,Y+50
    
    SPACES(I)=0
  Next I

  T=0 : Rem: turn
  Z=0 : Rem: zone
Return

CHECK_ZONE:
  For I=0 To 8
    If Z-1=I
      If SPACES(I)=0
        X1=(55+(I mod 3)*52)+5 : X2=X1+30
        Y1=(55+(I/3)*52)+5 : Y2=Y1+30
        Ink 2
        Draw X1,Y1 To X2,Y2
        Draw X1+30,Y1 To X2-30,Y2
        SPACES(I)=1
        T=1
      End If
    End If
  Next I
Return

OPPONENT_MOVE:
  Wait 15
  
  BEST_MOVE=-1

  For L=0 To 7
    P1=WIN(L,0) : P2=WIN(L,1) : P3=WIN(L,2)
    If SPACES(P1)=2 and SPACES(P2)=2 and SPACES(P3)=0 Then BEST_MOVE=P3 : Exit
    If SPACES(P1)=2 and SPACES(P3)=2 and SPACES(P2)=0 Then BEST_MOVE=P2 : Exit
    If SPACES(P2)=2 and SPACES(P3)=2 and SPACES(P1)=0 Then BEST_MOVE=P1 : Exit
  Next L

  If BEST_MOVE=-1
    For L=0 To 7
      P1=WIN(L,0) : P2=WIN(L,1) : P3=WIN(L,2)
      If SPACES(P1)=1 and SPACES(P2)=1 and SPACES(P3)=0
        BEST_MOVE=P3 : Exit
      End If
      If SPACES(P1)=1 and SPACES(P3)=1 and SPACES(P2)=0
        BEST_MOVE=P2 : Exit
      End If
      If SPACES(P2)=1 and SPACES(P3)=1 and SPACES(P1)=0
        BEST_MOVE=P1 : Exit
      End If
    Next L
  End If

  If BEST_MOVE=-1 and SPACES(4)=0
    BEST_MOVE=4
  End If

  If BEST_MOVE=-1
    Repeat
      I=Rnd(8)
    Until SPACES(I)=0
    BEST_MOVE=I
  End If

  I=BEST_MOVE
  SPACES(I)=2
  X=55+(I mod 3)*52
  Y=55+(I/3)*52
  CX=X+20
  CY=Y+20
  Ink 2 : Circle CX,CY,15
Return

CHECK_WIN:
  WINNER=0

  For L=0 To 7
    P1=WIN(L,0) : P2=WIN(L,1) : P3=WIN(L,2)
    If SPACES(P1)>0 and SPACES(P1)=SPACES(P2) and SPACES(P2)=SPACES(P3)
      WINNER=SPACES(P1)
    End If
  Next L

  If WINNER>0
    Ink 2 : Paper 0
    Locate 0,0 : Print "Game over! Player ";WINNER;" Wins!"
    Wait 100
    Locate 0,2 : Print "press any key"
    Wait Key
    Gosub SETUP_GAME
  End If

  If WINNER=0
    FULL=1
    For I=0 To 8
      If SPACES(I)=0
        FULL=0
      End If
    Next I
    If FULL=1
      Ink 2 : Paper 0
      Locate 0,0 : Print "It's a DRAW"
      Wait 100
      Locate 0,2 : Print "press any key"
      Wait Key
      Gosub SETUP_GAME
    End If
  End If
Return
